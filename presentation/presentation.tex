\section{Project Introduction \& Research Background}

\begin{frame}{NextKey Overview}{Redefining Vietnamese Mobile Typing}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{The Continuous Latin Input Paradigm}
        {\footnotesize Traditional IMEs (Telex/VNI) require typing explicit tone marks and accents. NextKey introduces a continuous Latin input paradigm (similar to Chinese Pinyin).}
      \end{block}
      \begin{exampleblock}{Three Core Technical Pillars}
        \begin{itemize}\setlength{\itemsep}{1.5pt}\footnotesize
          \item \textbf{Continuous Unaccented Input:}\\
            Type text without spaces or tone marks (e.g., \texttt{"toidihoc"} $\to$ \textbf{"toi di hoc"}).
          \item \textbf{End-to-End Deep Learning:}\\
            Jointly predicts word boundaries and restores diacritics in a single real-time forward pass.
          \item \textbf{100\% Edge-Based Execution:}\\
            Fully offline execution with latency $< 50\text{ ms}$ and absolute user data privacy.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{Ergonomic Efficiency}
        \textbf{\footnotesize Keystroke Savings (KS):}
        \vspace{-0.1cm}
        \begin{equation*}
          \text{KS} = \left(1 - \frac{|X_{\text{continuous}}|}{|X_{\text{Telex}}|}\right) \times 100\%
        \end{equation*}
        \vspace{-0.1cm}
        \begin{itemize}\setlength{\itemsep}{2pt}\footnotesize
          \item \textbf{20\%--40\% reduction} in screen taps.
          \item Eliminates explicit spacebar \& tone keys.
          \item Lowers thumb strain \& RSI injury risks.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Problem Formulation: Multi-Noise Transformation}{Joint Restoration of Compound Typing Errors}
  \begin{block}{Input-Output Transformation Task}
    \centering
    \textbf{Raw User Input ($X$):} \texttt{"toidihocbakk"} $\quad\xrightarrow{\quad\text{\textbf{NextKey Engine}}\quad}\quad$ \textbf{Target Output ($Y$):} \textbf{"toi di hoc bai"}
  \end{block}
  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Compound Noise Dimensions}
        \begin{itemize}\setlength{\itemsep}{2pt}\footnotesize
          \item \textbf{1. Diacritic Erasure:}\\
            Absence of all 6 tones \& 3 vowel marks.
          \item \textbf{2. Boundary Erasure:}\\
            Continuous typing without spaces.
          \item \textbf{3. QWERTY Spatial Typos:}\\
            Adjacent fat-finger key errors ($k \to i$).
          \item \textbf{4. Residual IME Codes:}\\
            Incomplete Telex codes (\texttt{"ddang"}).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Joint Multi-Task Objective}
        \begin{equation*}
          \mathcal{L}_{\text{total}} = \lambda_{\text{diac}} \mathcal{L}_{\text{diac}} + \lambda_{\text{seg}} \mathcal{L}_{\text{seg}} + \lambda_{\text{typo}} \mathcal{L}_{\text{typo}}
        \end{equation*}
        \begin{itemize}\setlength{\itemsep}{3pt}\footnotesize
          \item \textbf{$\mathcal{L}_{\text{diac}}$:} Diacritic tone restoration loss.
          \item \textbf{$\mathcal{L}_{\text{seg}}$:} Word boundary segmentation loss.
          \item \textbf{$\mathcal{L}_{\text{typo}}$:} Spatial typo correction loss.
          \item \textbf{Single-Pass:} Resolves all 3 tasks jointly.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Core Research Questions (RQ1 -- RQ6)}{Foundational Scientific Inquiries Driving NextKey}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Modeling \& Noise Inquiries}
        \begin{itemize}\setlength{\itemsep}{4pt}\footnotesize
          \item \textbf{RQ1 (Restoration Feasibility):}\\
            Can a compact network restore text from compound noise (accents, spaces, typos)?
          \item \textbf{RQ2 (Architecture Pareto Trade-off):}\\
            How do non-autoregressive taggers compare against Seq2Seq decoders in speed vs. accuracy?
          \item \textbf{RQ3 (Noise Calibration):}\\
            Does QWERTY spatial noise modeling improve real-world out-of-domain generalization?
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Edge Optimization \& UX Inquiries}
        \begin{itemize}\setlength{\itemsep}{4pt}\footnotesize
          \item \textbf{RQ4 (Quantization \& Distillation):}\\
            Can KD + INT8 quantization achieve $>95\%$ teacher parity with $<50\text{ms}$ latency?
          \item \textbf{RQ5 (Candidate Selection Utility):}\\
            Does Top-$k$ candidate generation improve practical typing Keystroke Savings (KS)?
          \item \textbf{RQ6 (Privacy-Preserving Edge):}\\
            Can on-device adaptation learn user vocabulary with zero remote server logging?
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
